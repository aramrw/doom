ACTOR Beholder
{
  Health 300
  Radius 40
  Height 40
  Mass 300
  Speed 8
  PainChance 112
  Monster
  +FLOAT
  +NOGRAVITY
  SeeSound "monster/sight"
  PainSound "monster/pain"
  DeathSound "monster/death"
  ActiveSound "monster/active"
  MeleeSound "monster/melee"
  Obituary "%o missed %p saving throw when attacked by a Beholder."
  HitObituary "%o was mangled by a Beholder's bite."
  BloodColor "BC CC 28"
  States
  {
  Spawn:
    BHLD ABCD 10 A_Look
    Loop
  See:
    BHLD AABBCCDD 3 A_Chase
    Loop
  Melee:
    BHLD E 4 A_FaceTarget
    BHLD F 6 A_CustomMeleeAttack (6,"monster/melee")
    Goto See
  Missile:
	BHLD G 4 A_FaceTarget
	BHLD G 0 A_Jump (128, "Missile2") //50% chance to jump to second attack
    BHLD G 8 A_SpawnProjectile("BhldPetrify",16,0)
    Goto See
  Missile2: //second attacks do not need to be named Missile2 that's just how Werly likes to do it
	BHLD G 0 A_FaceTarget
	BHLD G 8 A_SpawnProjectile ("BhldDisintegrate")
	Goto See
  Pain:
    BHLD H 0
    BHLD H 8 A_Pain
    Goto See
  Death:
    BHLD J 8
    BHLD K 8 A_Scream
    BHLD LM 8
    BHLD N 8 A_NoBlocking
    BHLD O -1 A_SetFloorClip
    Stop
  Raise:
    BHLD O 8 A_UnSetFloorClip
    BHLD NMLKJ 8
    Goto See
  }
}
