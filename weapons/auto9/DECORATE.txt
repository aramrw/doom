// AUTO-9 (SINGLE WEAPON MODE)

	Weapon.BobSpeed 2.0
	Weapon.BobStyle Alpha
	Weapon.KickBack 100
	Weapon.AmmoGive 20
	Weapon.AmmoType1 "Clip"
	Weapon.AmmoUse1 1
	Weapon.AmmoType2 "Clip"
	Weapon.AmmoUse2 1
	Weapon.UpSound "weapons/AUTO-9/ready1"

		AUT9 A 1 A_JumpIf(GetPlayerInput(INPUT_BUTTONS)&BT_ATTACK,"Hold1")
		Goto FireEnd

		AUT9 A 1 A_JumpIf(GetPlayerInput(INPUT_BUTTONS)&BT_ATTACK,"Hold4")
		Goto FireEnd
	Hold4:
		AUT9 B 0 A_JumpIfNoAmmo("FireEnd")
		AUT9 B 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_GunFlash; A_SetPitch(pitch-0.6); }
		AUT9 B 1 Bright A_SetPitch(pitch+0.6)
		AUT9 A 1 A_JumpIf(GetPlayerInput(INPUT_BUTTONS)&BT_ATTACK,"Hold5")
		Goto FireEnd
	Hold5:
		AUT9 C 0 A_JumpIfNoAmmo("FireEnd")
		AUT9 C 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_GunFlash; A_SetPitch(pitch-0.9); }
		AUT9 C 1 Bright A_SetPitch(pitch+0.9)
		AUT9 A 1
		Goto FireEnd
	FireEnd:
		AUT9 G 1 { A_StopSound(5); A_StartSound("weapons/AUTO-9/firestop",7); }
		AUT9 FFFFFFFFFG 1
		AUT9 A 1 Offset(0, 28)
		AUT9 A 1 Offset(0, 30)
		AUT9 A 1 Offset(0, 32)
		AUT9 AAAAA 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)

		AUT9 L 2 Offset(12, 18)
		AUT9 M 2 Offset(10, 20)

		AUT9 H 1 Offset(2, 28)

// AUTO-9 (DUAL-WIELD MODE)

ACTOR DualAUTO-9 : Weapon
{
	Weapon.SlotNumber 2
	Weapon.BobSpeed 2.0
	Weapon.BobStyle Alpha
	Weapon.BobRangeX 0.5
	Weapon.BobRangeY 0.5
	Weapon.KickBack 100
	Weapon.AmmoGive 20
	Weapon.AmmoType1 "Clip"
	Weapon.AmmoUse1 1
	Weapon.AmmoType2 "Clip"
	Weapon.AmmoUse2 1
	Weapon.UpSound "weapons/AUTO-9/ready3"
	Inventory.PickupMessage ""
	Obituary "%o was totally overdrilled by %k's Dual AUTO-9."
	Decal "BulletChip"
	Tag "Dual AUTO-9"
	Scale 0.4
	States
	{
	Ready:
		TNT1 A 1 A_WeaponReady
		Loop
	Select:
		DUA9 A 0 { A_Overlay(3,"IdleRight"); A_Overlay(-3,"IdleLeft"); }
		DUA9 A 1 A_Raise(12)
		Wait
	Deselect:
		DUA9 O 1 { A_ClearOverlays(-3,3); A_StopSound(5); A_StopSound(6); A_Lower(12); }
		Loop
	Fire:
		TNT1 AAAAAAAAAAAAAAAAAAAA 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
		Goto Ready
	IdleRight:
		DUA9 A 1
		{
		if( GetPlayerInput(INPUT_BUTTONS) & (BT_ALTATTACK) && CountInv("Clip")>0 )
			{ return state("FireRight"); }
		return state(""); 
		}
		Loop
	FireRight:
		DUA9 A 2
		DUA9 B 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_StartSound("weapons/AUTO-9/fireloop",5,CHANF_LOOPING); A_SetPitch(pitch-0.6); A_AlertMonsters; A_GunFlash; }
		DUA9 B 1 Bright A_SetPitch(pitch+0.6)
		DUA9 A 1 A_JumpIf(GetPlayerInput(INPUT_BUTTONS)&BT_ALTATTACK,"Hold1")
		Goto FireRightEnd
	Hold1:
		DUA9 C 0 A_JumpIfNoAmmo("FireRightEnd")
		DUA9 C 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_SetPitch(pitch-0.9); A_GunFlash; }
		DUA9 C 1 Bright A_SetPitch(pitch+0.9)
		DUA9 A 1 A_JumpIf(GetPlayerInput(INPUT_BUTTONS)&BT_ALTATTACK,"Hold2")
		Goto FireRightEnd
	Hold2:
		DUA9 D 0 A_JumpIfNoAmmo("FireRightEnd")
		DUA9 D 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_SetPitch(pitch-0.6); A_GunFlash; }
		DUA9 D 1 Bright A_SetPitch(pitch+0.6)
		DUA9 A 1 A_JumpIf(GetPlayerInput(INPUT_BUTTONS)&BT_ALTATTACK, "Hold3")
		Goto FireRightEnd
	Hold3:
		DUA9 E 0 A_JumpIfNoAmmo("FireRightEnd")
		DUA9 E 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_SetPitch(pitch-0.9); A_GunFlash; }
		DUA9 E 1 Bright A_SetPitch(pitch+0.9)
		DUA9 A 1 A_JumpIf(GetPlayerInput(INPUT_BUTTONS)&BT_ALTATTACK, "Hold4")
		Goto FireRightEnd
	Hold4:
		DUA9 B 0 A_JumpIfNoAmmo("FireRightEnd")
		DUA9 B 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_SetPitch(pitch-0.6); A_GunFlash; }
		DUA9 B 1 Bright A_SetPitch(pitch+0.6)
		DUA9 A 1 A_JumpIf(GetPlayerInput(INPUT_BUTTONS)&BT_ALTATTACK, "Hold5")
		Goto FireRightEnd
	Hold5:
		DUA9 C 0 A_JumpIfNoAmmo("FireRightEnd")
		DUA9 C 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_SetPitch(pitch-0.9); A_GunFlash; }
		DUA9 C 1 Bright A_SetPitch(pitch+0.9)
		DUA9 A 1
		Goto FireRightEnd
	FireRightEnd:
		DUA9 F 1 { A_StartSound("weapons/AUTO-9/firestop",7); A_StopSound(5); }
		DUA9 GGGGGGFAA 1
		DUA9 AAAAA 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
		Goto IdleRight
	AltFire:
		TNT1 AAAAAAAAAAAAAAAAAAAA 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
		Goto Ready
	IdleLeft:
		DUA9 N 1
		{
		if( GetPlayerInput(INPUT_BUTTONS) & (BT_ATTACK) && CountInv("Clip")>0 )
			{ return state("FireLeft"); }
		return state(""); 
		}
		Loop
	FireLeft:
		DUA9 N 2
		DUA9 H 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_StartSound("weapons/AUTO-9/fireloop",6,CHANF_LOOPING); A_SetPitch(pitch-0.6); A_AlertMonsters; A_GunFlash; }
		DUA9 H 1 Bright A_SetPitch(pitch+0.6)
		DUA9 N 1 A_JumpIf(GetPlayerInput(INPUT_BUTTONS)&BT_ATTACK, "AltHold1")
		Goto FireLeftEnd
	AltHold1:
		DUA9 I 0 A_JumpIfNoAmmo("FireLeftEnd")
		DUA9 I 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_SetPitch(pitch-0.9); A_GunFlash; }
		DUA9 I 1 Bright A_SetPitch(pitch+0.9)
		DUA9 N 1 A_JumpIf(GetPlayerInput(INPUT_BUTTONS)&BT_ATTACK, "AltHold2")
		Goto FireLeftEnd
	AltHold2:
		DUA9 J 0 A_JumpIfNoAmmo("FireLeftEnd")
		DUA9 J 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_SetPitch(pitch-0.6); A_GunFlash; }
		DUA9 J 1 Bright A_SetPitch(pitch+0.6)
		DUA9 N 1 A_JumpIf(GetPlayerInput(INPUT_BUTTONS)&BT_ATTACK, "AltHold3")
		Goto FireLeftEnd
	AltHold3:
		DUA9 K 0 A_JumpIfNoAmmo("FireLeftEnd")
		DUA9 K 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_SetPitch(pitch-0.9); A_GunFlash; }
		DUA9 K 1 Bright A_SetPitch(pitch+0.9)
		DUA9 N 1 A_JumpIf(GetPlayerInput(INPUT_BUTTONS)&BT_ATTACK, "AltHold4")
		Goto FireLeftEnd
	AltHold4:
		DUA9 H 0 A_JumpIfNoAmmo("FireLeftEnd")
		DUA9 H 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_SetPitch(pitch-0.6); A_GunFlash; }
		DUA9 H 1 Bright A_SetPitch(pitch+0.6)
		DUA9 N 1 A_JumpIf(GetPlayerInput(INPUT_BUTTONS)&BT_ATTACK, "AltHold5")
		Goto FireLeftEnd
	AltHold5:
		DUA9 I 0 A_JumpIfNoAmmo("FireLeftEnd")
		DUA9 I 1 Bright { A_FireBullets(1,1,-1,frandom(6,9),"BulletPuff"); A_SetPitch(pitch-0.9); A_GunFlash; }
		DUA9 I 1 Bright A_SetPitch(pitch+0.9)
		DUA9 N 1
		Goto FireLeftEnd
	FireLeftEnd:
		DUA9 L 1 { A_StartSound("weapons/AUTO-9/firestop",8); A_StopSound(6); }
		DUA9 MMMMMMLNN 1
		DUA9 NNNNN 1 A_WeaponReady(WRF_NOFIRE|WRF_NOSWITCH)
		Goto IdleLeft
	Flash:
		TNT1 A 1 Bright A_Light2
		TNT1 A 1 Bright A_Light1
		goto Lightdone
	Spawn:
		DUA9 Z -1 // IMPORTANT: For grabbing a single weapon before the second one, please use 'AUTO-9Weapon' actor for spawning!
		Stop
	}
}

ACTOR AUTO-9Weapon : CustomInventory // TIP: This actor can be used to pick up the weapons sequentially. Use it to enable dual-wield mode when you pickup a second AUTO-9!
{
	Scale 0.4
	Inventory.PickupSound "misc/w_pkup"
	Inventory.PickupMessage "You got the AUTO-9! Your move, creep!"
	States
	{
	Spawn:
		AUT9 Z -1
		Stop
	Pickup:
		TNT1 A 0 A_JumpIfInventory("AUTO-9", 1, "Pickup2")
		TNT1 A 0 A_GiveInventory("AUTO-9", 1)
		Stop
	Pickup2:
		TNT1 A 0 A_JumpIfInventory("DualAUTO-9", 1, 2)
		TNT1 A 0 A_Print("Now you can use two AUTO-9 at once! Watch your ammo, OK?")
		TNT1 A 0 A_GiveInventory("DualAUTO-9", 1)
		Stop
	}
}
