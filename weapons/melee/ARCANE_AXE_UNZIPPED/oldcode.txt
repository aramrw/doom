ACTOR QuakeAxe : Weapon 25032
{
   +MELEEWEAPON
   Inventory.PickupSound "weapons/quakeaxe"
   Inventory.PickupMessage "You got the axe!"
   Weapon.SlotNumber 1
   Scale 0.5
   States
   {
   Spawn:
      QAXP A -1
      Loop
   Ready:
      QAXE A 1 A_WeaponReady
      Loop
   Deselect:
      QAXE A 1 A_Lower
      Loop
   Select:
      QAXE A 1 A_Raise
      Loop
   Fire:
      QAXE B 4 A_PlaySound("weapons/quakeaxe",CHAN_WEAPON)
      QAXE C 4
      QAXE D 4 A_CustomPunch(12,0,0,"QAxePuff")
      QAXE E 4 A_ReFire
      Goto Ready
   }
}

Actor QAxePuff : BulletPuff
{
	+PUFFONACTORS
	States
	{
	Melee:
	Crash:
		PUFF A 4 Bright A_PlaySound("weapons/quakeaxe/hitsolid")
		PUFF BCD 4
		Stop
	XDeath:
		TNT1 A 1 A_PlaySound("weapons/quakeaxe/hitflesh")
		Stop
	}
}
